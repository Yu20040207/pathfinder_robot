cmake_minimum_required(VERSION 3.8)
project(fwr_rviz_nav)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(fwr_interfaces REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)

add_library(${PROJECT_NAME} SHARED src/two_point_nav_panel.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC include)
ament_target_dependencies(${PROJECT_NAME} rclcpp rclcpp_action rcl_interfaces rviz_common pluginlib geometry_msgs nav_msgs nav2_msgs fwr_interfaces std_srvs std_msgs visualization_msgs tf2 tf2_ros)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins_description.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_point_selection test/test_point_selection.cpp)
  target_include_directories(test_point_selection PRIVATE include)
  ament_add_gtest(test_navigation_route test/test_navigation_route.cpp)
  target_include_directories(test_navigation_route PRIVATE include)
  ament_target_dependencies(test_navigation_route nav2_msgs geometry_msgs)
  ament_add_gtest(test_real_nav_start_flow test/test_real_nav_start_flow.cpp)
  target_include_directories(test_real_nav_start_flow PRIVATE include)
endif()

ament_package()
